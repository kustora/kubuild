## Document AST Types (`@kubuild/schema`)

The `@kubuild/schema` package contains the zero-dependency canonical AST specifications. Every document authored in the editor, stored in database records, or streamed over APIs adheres to these schemas.

### `PageDocument`

The root container representing a complete `.stora` single-page document.

export interface PageDocument {
  schema: 'stora.page';
  version: string;
  metadata?: DocumentMetadata;

  theme?: Theme;
  document: RootPageNode;

import {
  isAssetReference,
  isVariableBinding,
  isActionBinding,
  isAnimationConfig,
  isProjectDocument,
} from '@kubuild/schema';

if (isVariableBinding(propValue)) {
  console.log('Resolving runtime variable:', propValue.key);
}

if (isAssetReference(propValue)) {
  console.log('Fetching asset:', propValue.assetId);
}

import {
  PageDocumentSchema,

const validDoc = PageDocumentSchema.parse(untrustedJson);
